# Ajax 실습

- 영화 순위 api
- 카카오 이미지 검색 api 활용

```jsx
async function myFunc() {
    const kobisUrl = 'http://www.kobis.or.kr/kobisopenapi/webservice/rest/boxoffice/searchDailyBoxOfficeList.json';
    const key = '545bac822b06f81c4d4b542aa93b3c2d';
    let date = $('#dateInput').val().replace(/-/g, '');
    console.log(date)

    $.ajax({
        async: true,                                   // 비동기화 여부 (default : true)
        url: kobisUrl,                                 // 요청할 서버 URL
        type: 'get',                                   // 타입 (get, post, put 등등)
        data: {
            key: key,
            targetDt: date,
        },
        dataType: 'json',                               // 서버가 보내주는 결과 데이터가 JSON(default: json)
        success: async function (result) {              // 결과 성공 콜백함수
            // console.log(result);
            for (const obj of result.boxOfficeResult.dailyBoxOfficeList) {
                let thumbnailUrl = await getPosterImg(obj.movieNm)
                console.log(thumbnailUrl)
                $('#myTable').append(
                    `<tr><td>${obj.rank}</td>` +
                    `<td><img src="${thumbnailUrl}" alt="이미지"></td>` +
                    `<td>${obj.movieNm}</td>` +
                    `<td>${obj.audiAcc}</td>` +
                    `<td>${obj.movieCd}</td>` +
                    `<td><input class="delete-btn" type="button" value="삭제"></td></tr>`);
            }
        },
        error: function (request, status, error) { // 결과 에러 콜백함수
            console.log(error);
            alert("호출 실패");
        }
    })

    $(document).on("click", ".delete-btn", function() {
        $(this).closest("tr").remove(); // 클릭된 버튼의 부모 <tr>을 삭제
    });
}
async function getPosterImg(movieName)  {
    const kakaoUrl = 'https://dapi.kakao.com/v2/search/image';
    const kakaoApiKey = 'KakaoAK 461917e887a47c9197f3b0a6d0fff96d'

    try {
        const result = await $.ajax({
            async: true,                                   // 비동기화 여부 (default : true)
            url: kakaoUrl,                                 // 요청할 서버 URL
            type: 'get',
            headers: {
                "Authorization": kakaoApiKey // API 키 포함
            },// 타입 (get, post, put 등등)
            data: {
                query: movieName + " 포스터"
            },
            dataType: 'json',                               // 서버가 보내주는 결과 데이터가 JSON(default: json)
            success: function (result) {              // 결과 성공 콜백함수
                // console.log(result);
                // console.log(result.documents[0].thumbnail_url);
            },
            error: function (request, status, error) { // 결과 에러 콜백함수
                console.log(error);
                alert("호출 실패");
            }
        })
        return result.documents[0].thumbnail_url;
    } catch (error) {
        console.log(error);
    }

}
```
