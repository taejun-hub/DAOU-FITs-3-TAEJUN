package javaexam3;

import java.util.StringTokenizer;

public class Test03 {
    public static void main (String[] args) {
        String str = "1.22,4.12,5.93,8.71,9.34";
        double[] data = new double[5];
        double sum = 0;

        StringTokenizer st = new StringTokenizer(str, ",");
        for (int i = 0; st.hasMoreElements(); i++) {
            data[i] = Double.parseDouble(st.nextToken());
        }
        for (double x : data) {
            sum += x;
        }
        System.out.println("합계: " + String.format("%.3f", sum));
        System.out.println("평균: " + String.format("%.3f", sum/5));
    }
}
```

        Random rand = new Random();
        for (int i = 1; i <= 10; i++) {
            map.put(i, rand.nextInt(10) + 1);
        }
        for (int value: map.values()) {
            System.out.print(value + " ");
        }
        System.out.println();

        double sum = 0;
        for (int key: map.keySet()) {
            sum += map.get(key);
        }
        System.out.println("합계: " + String.format("%.2f", sum));
        System.out.println("평균: " + String.format("%.2f", sum / 10));


    }

package javaexam3.plane;

public abstract class Plane {
    private String planeName;
    private int fuelSize;

    public Plane() {}
    public Plane(String planeName, int fuelSize) {
        this.planeName = planeName;
        this.fuelSize = fuelSize;
    }

    public String getPlaneName() {
        return this.planeName;
    }

    public int getFuelSize() {
        return this.fuelSize;
    }

    public void setPlaneName(String planeName) {
        this.planeName = planeName;
    }

    public void setFuelSize(int fuelSize) {
        this.fuelSize = fuelSize;
    }

    public void refuel(int fuel) {
        this.fuelSize += fuel;
    }

    public abstract void flight(int distance);

    public void printPlaneInfo() {
        System.out.println(this.planeName + "     " + this.fuelSize);
    }
}
```

package javaexam3.company;

public abstract class Employee {
    private String name;
    private int number;
    private String department;
    private int salary;

    public Employee() {}
    public Employee(String name, int number, String department, int salary) {
        this.name = name;
        this.number = number;
        this.department = department;
        this.salary = salary;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public int getNumber() {
        return number;
    }

    public void setNumber(int number) {
        this.number = number;
    }

    public String getDepartment() {
        return department;
    }

    public void setDepartment(String department) {
        this.department = department;
    }

    public int getSalary() {
        return this.salary;
    }

    public void setSalary(int salary) {
        this.salary = salary;
    }


    public String getEmployeeInfo() {
        return this.getName() + " " + this.getDepartment() + " " + this.getSalary();
    }
    public abstract double tax();
}
