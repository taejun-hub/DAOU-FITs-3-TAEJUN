# 실습 과제 (1/14)

**1. 테이블 생성한 "명령어 스크립트 파일"**
```
CREATE TABLE CUSTOMERS (
    CUSTOMER_ID NUMBER PRIMARY KEY,
    NAME VARCHAR2(50) NOT NULL,
    EMAIL VARCHAR2(100) UNIQUE,
    GENDER VARCHAR2(50),
    ASSET NUMBER,
    RATING VARCHAR2(50),
    CREATED_AT DATE DEFAULT SYSDATE
);
```
```
CREATE TABLE LOAN_PRODUCT (
    LOAN_PRODUCT_ID NUMBER PRIMARY KEY,
    NAME VARCHAR2(50) NOT NULL,
    COMPANY_ID VARCHAR2(100) UNIQUE,
    HAS_PREFERENTIAL_RATE CHAR(1),
    ASSET NUMBER,
    RATING VARCHAR2(50),
    AVERAGE_INTEREST_RATE FLOAT,
    MAXIMUM_LIMIT NUMBER,
    RELEASE_DATE DATE,
    CREATED_AT DATE DEFAULT SYSDATE
);

```
```
CREATE TABLE REVIEWS (
    REVIEW_ID NUMBER PRIMARY KEY,
    REVIEWER_ID NUMBER NOT NULL,
    CONSTRAINT FK_COSTOMER_ID FOREIGN KEY(REVIEWER_ID) REFERENCES CUSTOMERS(CUSTOMER_ID),
    LOAN_PRODUCT_ID NUMBER NOT NULL,
    CONSTRAINT FK_LOAN_PRODUCT_ID FOREIGN KEY(LOAN_PRODUCT_ID) REFERENCES LOAN_PRODUCT(LOAN_PRODUCT_ID),
    TITLE VARCHAR2(50),
    CONTENT LONG,
    VIEWS NUMBER,
    COMMENTS NUMBER,
    CREATED_AT DATE DEFAULT SYSDATE
);

```
