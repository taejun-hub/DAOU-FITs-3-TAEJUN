## 3. 샘플 데이터 삽입
```sql
-- ORDERS CREATE
CREATE TABLE orders (
    order_id VARCHAR2(64) NOT NULL,
    order_date DATE DEFAULT SYSDATE NOT NULL,
    account_id	VARCHAR(64)	NOT NULL,
		stock_id	VARCHAR(64)	NOT NULL,
    status VARCHAR2(10),
    type VARCHAR2(4),
    quantity NUMBER,
    price NUMBER,
    traded NUMBER,
    not_traded NUMBER,
    created_at TIMESTAMP DEFAULT SYSTIMESTAMP,
    updated_at TIMESTAMP DEFAULT SYSTIMESTAMP
);

-- PK 설정
ALTER TABLE ORDERS ADD CONSTRAINT PK_ORDERS PRIMARY KEY (
		order_id,
		order_date
);

-- 샘플 데이터 INSERT
INSERT INTO orders (order_id, account_id, stock_id, status, type, quantity, price)
SELECT 
    'order' || LPAD(LEVEL, 3, '0') AS order_id,
    'ACCT' || LPAD(TRUNC(DBMS_RANDOM.VALUE(1, 100)), 4, '0') AS account_id,
    'STOCK' || LPAD(TRUNC(DBMS_RANDOM.VALUE(1, 100)), 4, '0') AS stock_id,-- 랜덤한 주식 선택
    CASE WHEN MOD(LEVEL, 2) = 0 THEN 'OPEN' ELSE 'CLOSED' END AS status, -- 상태 (OPEN / CLOSED)
    CASE WHEN MOD(LEVEL, 3) = 0 THEN 'BUY' ELSE 'SELL' END AS type, -- 거래 유형 (BUY / SELL)
    TRUNC(DBMS_RANDOM.VALUE(1, 1000)) AS quantity, -- 1~1000 랜덤 수량
    ROUND(DBMS_RANDOM.VALUE(10, 500), 2) AS price -- 10 ~ 500 랜덤 가격
FROM 
    DUAL
CONNECT BY LEVEL <= 300;
```

```sql

-- TRANSACTIONS Table
CREATE TABLE TRANSACTIONS (
		transaction_id	VARCHAR(64)	NOT NULL,
		transaction_date DATE DEFAULT SYSDATE NOT NULL,
		order_id	VARCHAR(64)	NULL,
	  order_date DATE  NOT NULL,
		quantity NUMBER		NULL,
		price	NUMBER		NULL,
		change	NUMBER		NULL,
	  created_at TIMESTAMP DEFAULT SYSTIMESTAMP,
	  updated_at TIMESTAMP DEFAULT SYSTIMESTAMP
);

-- PK 설정
ALTER TABLE TRANSACTIONS ADD CONSTRAINT PK_TRANSACTIONS PRIMARY KEY (
		transaction_id,
		transaction_date
);

INSERT INTO transactions (transaction_id, transaction_date, order_id, order_date, quantity, price, change)
SELECT 
    'TRX020' as transaction_id,
    order_date + DBMS_RANDOM.VALUE(0, SYSDATE - order_date) AS transaction_date, -- order_date와 SYSDATE 사이 랜덤 값
    order_id as order_id,  -- 주문 ID
    order_date as order_date,
    128 AS quantity, -- 거래된 수량 (traded와 동일)
    price as price, -- 주문 가격
    0 AS change -- 가격 변동 (랜덤한 ±10 변동)
FROM 
    orders
WHERE order_id = 'order010';  -- 거래된 수량이 있는 주문만 선택
```
