# 실습과제

## 1. BookSearchCommand
```java
import java.sql.*;
import java.util.Scanner;

public class BookSearchCommand {
    public static void main(String[] args) {
        Connection conn = null;
        PreparedStatement pstmt = null;
        ResultSet rs = null;

        String JDBC_URL = "jdbc:oracle:thin:@localhost:1521:xe";
        String ID = "";
        String PW = "";

        try {
            // 1. Driver load
            Class.forName("oracle.jdbc.driver.OracleDriver");
            System.out.println("Driver loaded");

            // 2. DB Connection
            conn = DriverManager.getConnection(JDBC_URL, ID, PW);
            System.out.println("Connection established");

            // 사용자 input
            Scanner sc = new Scanner(System.in);
            System.out.println("Enter the keyword of book title: ");
            String keyword = sc.nextLine();

            // 3. Statement 생성
            // IN Parameter 를 이용해서 SQL query 를 작성
            String sql = "select bisbn, btitle, bauthor, bprice ";
            sql += "from book where btitle like ?";
            pstmt = conn.prepareStatement(sql);
            pstmt.setString(1, "%" + keyword + "%");

            //4. Query 실행
            rs = pstmt.executeQuery(); // select 구문이므로 executeQuery 사용
            
            //5. 결과 처리 
            while (rs.next()) {
                String tmp = "";
                tmp += rs.getString("bisbn") + " ";
                tmp += rs.getString("btitle") + " ";
                tmp += rs.getString("bauthor") + " ";
                tmp += rs.getString("bprice") + " ";
                System.out.println(tmp);
            }

        } catch (SQLException e) {
            // db connection 오류
            System.out.println("SQLException: " + e.getMessage());
        } catch (ClassNotFoundException e) {
            // jdbc driver class 오류
            System.out.println("ClassNotFoundException: " + e.getMessage());
        } finally {
            // 6. 리소스 해제
            try {
                if (rs != null) {
                    rs.close();
                }
                if (pstmt != null) {
                    pstmt.close();
                }
                if (conn != null) {
                    conn.close();
                }
            } catch (SQLException e) {
                System.out.println("could not close connection");
            }
        }
    }
}
```

## 2. Transaction 실습
```java
import java.sql.*;

public class TransactionTest {
    public static void main(String[] args) {
        Connection conn = null;
        PreparedStatement pstmt = null;
        String JDBC_URL = "jdbc:oracle:thin:@localhost:1521:xe";
        String ID = "";
        String PW = "";
        try {
            Class.forName("oracle.jdbc.driver.OracleDriver");

            conn = DriverManager.getConnection(JDBC_URL, ID, PW);
            // 기본적으로 autoCommit mode 로 Connection 생성
            // 만약 transaction을 설정하려면
            conn.setAutoCommit(false); // transaction 시작
            // 명시적으로 conn.commit() 혹은 conn.rollback() 이 호출되면 transaction 종료
            // 추가적으로 만약 해당 구문이 나오지 않고 connection 정상 종료되면 conn.commit() 으로 간주

            String sql = "insert into membertbl values(?, ?)";
            pstmt = conn.prepareStatement(sql);
            pstmt.setString(1, "2");
            pstmt.setString(2, "신사임당");
            int result = pstmt.executeUpdate(); // 리턴값은 영향 받은 row 의 수

            if (result == 1) {
                System.out.println("insert success");
            } else {
                System.out.println("insert fail");
            }
            conn.commit();

        } catch(SQLException e) {
            System.out.println("SQLException: " + e.getMessage());
        } catch (ClassNotFoundException e) {
            System.out.println("ClassNotFoundException: " + e.getMessage());
        } finally {
            try {
                pstmt.close();
                conn.close();
            } catch (SQLException e) {
                System.out.println("SQLException: " + e.getMessage());
            }
        }
    }
}
```
